# The questions that a typical CMake project asks about the compiler. The answers go to summary.txt.
cmake_minimum_required(VERSION 3.20)
project(fixture C)

include(CheckCCompilerFlag)
include(CheckCSourceCompiles)
include(CheckIncludeFile)
include(CheckLibraryExists)
include(CheckPIESupported)
include(CheckSymbolExists)
include(CheckTypeSize)
include(CheckIPOSupported)

foreach(h sys/epoll.h threads.h stdatomic.h sys/random.h valgrind/valgrind.h)
	string(MAKE_C_IDENTIFIER "HAVE_${h}" v)
	string(TOUPPER "${v}" v)
	check_include_file("${h}" ${v})
	list(APPEND checks ${v})
endforeach()

set(CMAKE_REQUIRED_DEFINITIONS -D_GNU_SOURCE)
check_symbol_exists(memfd_create sys/mman.h HAVE_MEMFD_CREATE)
check_symbol_exists(getrandom sys/random.h HAVE_GETRANDOM)
check_symbol_exists(strlcpy string.h HAVE_STRLCPY)
check_symbol_exists(reallocarray stdlib.h HAVE_REALLOCARRAY)
unset(CMAKE_REQUIRED_DEFINITIONS)
list(APPEND checks HAVE_MEMFD_CREATE HAVE_GETRANDOM HAVE_STRLCPY HAVE_REALLOCARRAY)

check_type_size(long SIZEOF_LONG)
check_type_size("long double" SIZEOF_LONG_DOUBLE)
check_type_size(__int128 SIZEOF_INT128)
list(APPEND checks SIZEOF_LONG SIZEOF_LONG_DOUBLE SIZEOF_INT128)

check_library_exists(m cos "" HAVE_LIBM)
list(APPEND checks HAVE_LIBM)

set(n 0)
foreach(f -Wall -Wextra -Wformat=2 -Wno-unused-parameter -Wimplicit-fallthrough -Wthis-is-not-a-warning -fstack-protector-strong -fstack-clash-protection -fcf-protection -fvisibility=hidden -ffunction-sections -fno-plt -fno-semantic-interposition -fno-strict-aliasing -flto -fsanitize=address -fopenmp -march=native -std=gnu17 -std=c23)
	math(EXPR n "${n} + 1")
	check_c_compiler_flag("${f}" FLAG_${n})
	list(APPEND flags "${f}=${FLAG_${n}}")
endforeach()

check_c_source_compiles("__attribute__((visibility(\"hidden\"))) int f(void) { return 0; } int main(void) { return f(); }" HAVE_VISIBILITY)
check_c_source_compiles("int main(int c, char **v) { (void)v; return __builtin_expect(c, 1) ? 0 : 1; }" HAVE_BUILTIN_EXPECT)
check_c_source_compiles("#include <stdatomic.h>\nint main(void) { _Atomic long x = 0; return (int)atomic_fetch_add(&x, 1); }" HAVE_C11_ATOMICS)
check_c_source_compiles("int main(void) { unsigned __int128 x = 1; return (int)(x >> 64); }" HAVE_INT128)
check_c_source_compiles("static int a(void) { return 0; } static int (*r(void))(void) { return a; } int f(void) __attribute__((ifunc(\"r\"))); int main(void) { return f(); }" HAVE_IFUNC)
list(APPEND checks HAVE_VISIBILITY HAVE_BUILTIN_EXPECT HAVE_C11_ATOMICS HAVE_INT128 HAVE_IFUNC)

check_pie_supported(OUTPUT_VARIABLE pie_out LANGUAGES C)
check_ipo_supported(RESULT HAVE_IPO OUTPUT ipo_out LANGUAGES C)
list(APPEND checks HAVE_IPO)

set(THREADS_PREFER_PTHREAD_FLAG ON)
find_package(Threads REQUIRED)

add_library(fix SHARED fix.c)
target_link_libraries(fix PRIVATE m)
add_executable(main main.c)
target_link_libraries(main PRIVATE fix Threads::Threads)

set(out "${CMAKE_BINARY_DIR}/summary.txt")
file(WRITE "${out}" "")
foreach(v CMAKE_C_COMPILER_ID CMAKE_C_COMPILER_VERSION CMAKE_C_COMPILER_FRONTEND_VARIANT CMAKE_C_SIMULATE_ID CMAKE_C_STANDARD_COMPUTED_DEFAULT CMAKE_C_EXTENSIONS_COMPUTED_DEFAULT CMAKE_C_COMPILE_FEATURES CMAKE_C_COMPILER_ABI CMAKE_C_SIZEOF_DATA_PTR CMAKE_C_BYTE_ORDER CMAKE_C_LIBRARY_ARCHITECTURE CMAKE_LIBRARY_ARCHITECTURE CMAKE_C_IMPLICIT_INCLUDE_DIRECTORIES CMAKE_C_IMPLICIT_LINK_DIRECTORIES CMAKE_C_IMPLICIT_LINK_LIBRARIES CMAKE_C_COMPILER_LINKER_ID CMAKE_C_COMPILER_AR CMAKE_C_COMPILER_RANLIB CMAKE_C_COMPILE_OPTIONS_PIC CMAKE_C_COMPILE_OPTIONS_PIE CMAKE_C_LINK_PIE_SUPPORTED CMAKE_C_LINK_NO_PIE_SUPPORTED CMAKE_THREAD_LIBS_INIT CMAKE_HAVE_LIBC_PTHREAD ${checks})
	file(APPEND "${out}" "${v}=${${v}}\n")
endforeach()
foreach(f ${flags})
	file(APPEND "${out}" "flag ${f}\n")
endforeach()
