# The known differences between rucc and the reference GCC, each with the rucc issue that owns it.
# The columns are globs: row, flag set, key. check takes the first line that matches.
#
# row       set         key                                   issue

# The kernel flag sets: no FPU on x86_64 and aarch64, the riscv64 kernel -march, and the macros that follow.
*           kernel      *                                     tamnd/rucc#4096

# -m32 on the x86_64 targets.
# GCC gives the FreeBSD and NetBSD i386 defaults: 53-bit long double on FreeBSD, x87 math (FLT_EVAL_METHOD 2), -march=i386 without CAS, MMX or SSE.
# rucc gives the i686 SSE defaults of Linux.
*           m32*        *                                     tamnd/rucc#4088
*           boot-i386   *                                     tamnd/rucc#4088

# Macros that still differ from the GCC of the persona: __bf16, the decimal float macros, and the _BitInt width.
# rucc gives __BITINT_MAXWIDTH__ 128, the widest _BitInt it compiles, where GCC gives 65535.
*           *           __BFLT16_*                            tamnd/rucc#4088
*           *           __DEC*                                tamnd/rucc#4088
*           *           __BITINT_MAXWIDTH__                   tamnd/rucc#4088

# x86_64 and aarch64: rucc does not have __builtin_speculation_safe_value.
*           *           __HAVE_SPECULATION_SAFE_VALUE         tamnd/rucc#4088

# aarch64: the ACLE macros, their spelling (65 and 14, not 'A' and 0xe), SME keywords and 16-byte CAS.
*-aarch64   *           __ARM_*                               tamnd/rucc#4088
*-aarch64   *           __arm_*                               tamnd/rucc#4088
*-aarch64   *           __GCC_HAVE_SYNC_COMPARE_AND_SWAP_16   tamnd/rucc#4088
*-aarch64   *           __FP_FAST_FMA*                        tamnd/rucc#4088

# riscv64: the BSD GCC defaults are rv64gc with -mcmodel=medany. rucc does not give the extension macros.
*-riscv64   *           __riscv_*                             tamnd/rucc#4088
*-riscv64   *           __FP_FAST_FMA*                        tamnd/rucc#4088
*-riscv64   *           __GCC_ATOMIC_*_LOCK_FREE              tamnd/rucc#4088
*-riscv64   *           __GCC_HAVE_SYNC_COMPARE_AND_SWAP_*    tamnd/rucc#4088
