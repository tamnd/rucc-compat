# The known differences between rucc and the reference GCC, each with the rucc issue that owns it.
# The columns are globs: row, flag set, key. check takes the first line that matches.
#
# row       set         key                                   issue

# -m32 on the x86_64 targets.
# GCC defines _Float16 there, and from GCC 14 also __bf16. rucc has neither type on 32 bit x86.
# GCC for FreeBSD makes _Float64x the binary128 format there. rucc makes it the format of long double.
*-x86_64    m32*        __FLT16_*                             tamnd/rucc#4324
*-x86_64    boot-i386   __FLT16_*                             tamnd/rucc#4324
*-x86_64    m32*        __BFLT16_*                            tamnd/rucc#4324
*-x86_64    boot-i386   __BFLT16_*                            tamnd/rucc#4324
freebsd-*   m32*        __FLT64X_*                            tamnd/rucc#4325
freebsd-*   boot-i386   __FLT64X_*                            tamnd/rucc#4325

# Macros that still differ from the GCC of the persona: __bf16 on riscv64 and the _BitInt width.
# GCC 15 on FreeBSD riscv64 has __bf16 and rucc does not, because rucc cannot move a half there yet.
# rucc gives __BITINT_MAXWIDTH__ 128, the widest _BitInt it compiles, where GCC gives 65535.
*-riscv64   *           __BFLT16_*                            tamnd/rucc#4276
*           *           __BITINT_MAXWIDTH__                   tamnd/rucc#2490

# aarch64: __ARM_NEON_SVE_BRIDGE, which promises <arm_neon_sve_bridge.h> and the SVE types.
*-aarch64   *           __ARM_NEON_SVE_BRIDGE                 tamnd/rucc#4225
