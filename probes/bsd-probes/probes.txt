# The probes. Each line is one run of the compiler.
# The columns are: name, system, arch, input, reads, arguments.
# The input is what the run gets on standard input:
#   none      nothing
#   ccv       the line __FreeBSD_cc_version, as bsd.compiler.mk sends it
#   features  the feature tests of rb probes (__has_extension, __has_feature, __has_builtin, __has_attribute)
#   unit      a one-line C unit, compiled with -c -o /dev/null -x c -
# The reads column says what the build reads from the run:
#   identity  COMPILER_TYPE and COMPILER_VERSION, as bsd.compiler.mk computes them from --version, and -v when --version names no compiler
#   vline     the word after "gcc version" in the -v output
#   line      the exit status and the first line of standard output
#   base      the exit status and the last path part of the first line of standard output
#   lines     the exit status and the lines of standard output that are not line markers
#   macros    the macros of macros.txt in the -dM output
#   status    the exit status only
#
# name                      system   arch     input     reads     arguments

# bsd.compiler.mk of FreeBSD.
